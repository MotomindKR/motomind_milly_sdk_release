cmake_minimum_required(VERSION 3.16)
project(milly_sdk_description)
find_package(ament_cmake REQUIRED)
# Release/development SDK root. Installed assets
# are self-contained and never depend on a developer's absolute path.
set(MILLY_DESCRIPTION_DIR "" CACHE PATH "Source milly_description containing meshes")
if(NOT MILLY_DESCRIPTION_DIR)
  foreach(candidate
      "${CMAKE_CURRENT_SOURCE_DIR}/../../../milly_description")
    if(EXISTS "${candidate}/meshes/J1.stl")
      get_filename_component(MILLY_DESCRIPTION_DIR "${candidate}" ABSOLUTE)
      break()
    endif()
  endforeach()
endif()
if(NOT EXISTS "${MILLY_DESCRIPTION_DIR}/meshes/J1.stl")
  message(FATAL_ERROR "Missing meshes. Set -DMILLY_DESCRIPTION_DIR=/path/to/milly_description")
endif()
set(MILLY_URDF "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/motomind_milly/robot/MILLY.urdf")
if(NOT EXISTS "${MILLY_URDF}")
  set(MILLY_URDF "${MILLY_DESCRIPTION_DIR}/urdf/MILLY.urdf")
endif()
if(NOT EXISTS "${MILLY_URDF}")
  message(FATAL_ERROR "Missing bundled SDK MILLY.urdf")
endif()
install(FILES "${MILLY_URDF}" DESTINATION share/${PROJECT_NAME}/milly_description/urdf)
install(DIRECTORY "${MILLY_DESCRIPTION_DIR}/meshes" DESTINATION share/${PROJECT_NAME}/milly_description)
# Retire the previous control launch from an existing colcon install.
install(CODE [[file(REMOVE "$ENV{DESTDIR}${CMAKE_INSTALL_PREFIX}/share/milly_sdk_description/launch/bringup.launch.py")]])
install(DIRECTORY launch rviz DESTINATION share/${PROJECT_NAME})
ament_package()
